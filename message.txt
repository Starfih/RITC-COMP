import requests
import numpy as np
import pandas as pd
from bs4 import BeautifulSoup
from time import sleep
import matplotlib.pyplot as plt

# ========= CONFIG =========
API = "http://localhost:9999/v1"
API_KEY = "Rotman"
HDRS = {"X-API-key": API_KEY}

NGN, WHEL, GEAR, RSM1000 = "NGN", "WHEL", "GEAR", "RSM1000"


entry_price = {NGN: None, WHEL: None, GEAR: None}

FEE_MKT = 0.01          # $/share (market)
ORDER_SIZE      = 5000
MAX_TRADE_SIZE  = 10_000
GROSS_LIMIT_SH  = 500_000
NET_LIMIT_SH    = 100_000
ENTRY_BAND_PCT  = 0.10   # enter if |div| > 0.50%
EXIT_BAND_PCT   = -0.1   # flatten if |div| < 0.20%
SLEEP_SEC       = 0.25
PRINT_HEARTBEAT = True

# ========= SESSION =========
s = requests.Session()
s.headers.update(HDRS)

# ========= BASIC HELPERS =========
def get_tick_status():
    r = s.get(f"{API}/case"); r.raise_for_status()
    j = r.json()
    return j["tick"], j["status"]

def best_bid_ask(ticker):
    r = s.get(f"{API}/securities/book", params={"ticker": ticker}); r.raise_for_status()
    book = r.json()
    bid = float(book["bids"][0]["price"]) if book["bids"] else 0.0
    ask = float(book["asks"][0]["price"]) if book["asks"] else 1e12
    return bid, ask

def mid_price(ticker):
    bid, ask = best_bid_ask(ticker)
    if bid == 0.0 and ask == 1e12:
        return None
    return 0.5 * (bid + ask)

def positions_map(call_type):
    r = s.get(f"{API}/securities"); r.raise_for_status()
    out = {p["ticker"]: int(p.get("position", 0)) for p in r.json()}
    unrealized = {p["ticker"]: int(p.get("unrealized", 0)) for p in r.json()}
    vwap = {p["ticker"]: int(p.get("vwap", 0)) for p in r.json()}
    for k in (NGN, WHEL, GEAR, RSM1000):
        out.setdefault(k, 0)
        unrealized.setdefault(k, 0)
        vwap.setdefault(k, 0)
    if call_type == 'unrealized':
        return unrealized
    if call_type == 'out':
        return out
    if call_type == 'vwap':
        return vwap

def place_mkt(ticker, action, qty):
    qty = int(max(1, min(qty, MAX_TRADE_SIZE)))
    r = s.post(f"{API}/orders",
               params={"ticker": ticker, "type": "MARKET",
                       "quantity": qty, "action": action})
    if PRINT_HEARTBEAT:
        print(f"ORDER {action} {qty} {ticker} -> {'OK' if r.ok else 'FAIL'}")
    return r.ok

def within_limits():
    pos = positions_map()
    gross = abs(pos[NGN]) + abs(pos[WHEL]) + abs(pos[GEAR])
    net   = pos[NGN] + pos[WHEL] + pos[GEAR]
    return ((gross) < GROSS_LIMIT_SH) and (abs(net) < NET_LIMIT_SH)

# ========= HISTORICAL (tables + betas) =========
def load_historical():
    r = s.get(f"{API}/news"); r.raise_for_status()
    news = r.json()
    if not news:
        print("No news yet. Start the case and ensure table is published.")
        return None
    soup = BeautifulSoup(news[0].get("body",""), "html.parser")
    table = soup.find("table")
    if not table:
        print("No <table> in news body.")
        return None

    rows = []
    for tr in table.find_all("tr"):
        cols = [td.get_text(strip=True) for td in tr.find_all("td")]
        if len(cols) == 5:
            rows.append(cols)

    df_hist = pd.DataFrame(rows[1:], columns=rows[0])
    df_hist["Tick"] = df_hist["Tick"].astype(int)
    for c in ["RSM1000", "NGN", "WHEL", "GEAR"]:
        df_hist[c] = df_hist[c].astype(float)
    return df_hist

def print_three_tables_and_betas(df_hist):
    # 1) Historical price table
    pd.set_option("display.float_format", lambda x: f"{x:0.6f}")
    print("\nHistorical Price Data:\n")
    print(df_hist.to_string(index=False))

    # 2) Correlation on tick returns
    returns = df_hist[["RSM1000", "NGN", "WHEL", "GEAR"]].pct_change().dropna()
    corr = returns.corr()
    print("\nHistorical Correlation:\n")
    print(corr.to_string())

    # 3) Volatility & beta (vs RSM1000)
    tick_vol = returns.std()
    idx_var  = returns["RSM1000"].var()
    beta_map = {t: float(np.cov(returns[t], returns["RSM1000"])[0,1] / idx_var)
                for t in ["RSM1000","NGN","WHEL","GEAR"]}
    vol_beta_df = pd.DataFrame({
        "Tick Volatility": tick_vol,
        "Beta vs RSM1000": [beta_map[t] for t in tick_vol.index]
    })
    print("\nHistorical Volatility and Beta:\n")
    print(vol_beta_df.to_string())
    return beta_map

def init_live_plot():

    line_whel, = ax.plot([], [], label="WHEL")

    ax.set_xlabel("Tick")
    ax.set_ylabel("Divergence (%)")
    ax.grid(True)
    ax.legend()
    fig.canvas.draw()
    fig.canvas.flush_events()
    return fig, ax, line_ngn, line_whel, line_gear

def update_live_plot(ax, line_ngn, line_whel, line_gear, ticks, series_ngn, series_whel, series_gear):
    # update data for all three lines
    line_ngn.set_data(ticks, series_ngn)
    line_whel.set_data(ticks, series_whel)
    line_gear.set_data(ticks, series_gear)
    # rescale axes
    ax.relim()
    ax.autoscale_view()
    plt.pause(0.01)  # let GUI process events



# ========= MAIN =========
def main():



    ma_list_s = {NGN: [mid_price(NGN)]*20, WHEL: [mid_price(WHEL)]*20, GEAR:[mid_price(GEAR)]*20}  # list of historical moving averages, long
    ma_list_l = {NGN: [mid_price(NGN)]*20, WHEL: [mid_price(WHEL)]*20, GEAR:[mid_price(GEAR)]*20}
    ma_list_d = {NGN: [mid_price(NGN)]*20, WHEL: [mid_price(WHEL)]*20, GEAR:[mid_price(GEAR)]*20}
    ma_s = {NGN: 0, WHEL: 0, GEAR: 0}  # moving averages
    ma_l = {NGN: 0, WHEL: 0, GEAR: 0}  # moving averages
    ma_d = {NGN: 0, WHEL: 0, GEAR: 0}
    hist = {NGN: [], WHEL: [], GEAR: []}  # list of historical stock prices
    days_s =  20  # Amount of days short moving average is calculated on
    days_l = 30 #Amount of days long moving average is calculated on
    ticks = 0
    growth_threshold_up = 0.005 # Average difference in moving average required to justify trade
    growth_threshold_down = -0.005 # Average difference in moving average required to justify trade
    previous_tick = -1
    last_order = {NGN: "NONE",WHEL: "NONE", GEAR: "NONE"}

    def moving_avg(stk, ma_yest, days):
        price = mid_price(stk)
        if price is None:
            return ma_yest  # don't break if book empty

        alpha = 2 / (days + 1)  # smoothing factor
        return alpha * price + (1 - alpha) * ma_yest
    

    def order_size(stk, delta):

        base = 9000
        scalar = 12000
        max = 10000

        size = int(base + scalar * abs(delta))
        if size > max:
            return max

        return size

    def profit_take(stk):
        # Get full portfolio dictionary
      

        position = positions_map("out")[stk]      # int
        unrealized =  positions_map("unrealized")[stk]



        if position == 0 or unrealized is None:
            return

        # Profit-taking threshold
        target_gain = 0.3   # 30% return
        target_loss = -0.2   # 20% loss ROI

        # Actual return
        pnl_pct = unrealized/abs(position)


        # If profit % exceeds threshold → exit
        if pnl_pct >= target_gain:

            if position > 0:
                place_mkt(stk, "SELL", abs(position))
                print('profittaken')
         

            elif position < 0:
                place_mkt(stk, "BUY", abs(position))
                print('profittaken')

        if pnl_pct <= target_loss:

            if position > 0:
                place_mkt(stk, "SELL", abs(position))
                print('proflossn')
         

            elif position < 0:
                place_mkt(stk, "BUY", abs(position))
                print('profitloss')

        


    # Run while case active
    tick, status = get_tick_status()
    while status == "ACTIVE":
        # current mids
        mid_idx = mid_price(RSM1000)
        mid_ngn = mid_price(NGN)
        mid_whe = mid_price(WHEL)
        mid_ger = mid_price(GEAR)

        stock_mid = {NGN: mid_ngn, WHEL: mid_whe, GEAR: mid_ger}  # mid bid/ask price of stocks

        for i in [NGN, WHEL, GEAR]:

            hist[i].append(stock_mid[i])

            ma_s[i] = moving_avg(i, ma_list_s[i][-1], days_s)
            ma_l[i] = moving_avg(i, ma_list_l[i][-1], days_l)
            ma_d[i] = moving_avg(i, ma_list_d[i][-1], 10)

            ma_list_s[i].append(ma_s[i])
            ma_list_l[i].append(ma_l[i])
            ma_list_d[i].append(ma_d[i])
           
            
            
            ma_delta = ma_s[i]-ma_l[i]


            upper_band = 0 
            lower_band = 0
            window = 10
            

            if len(ma_list_d[i]) >= window:
                data = ma_list_d[i][-window:] 

                mean = np.mean(data)
                stdev = np.std(data)

                upper_band = mean + 2 * stdev
                lower_band = mean - 2 * stdev

            
          
            def trade(i):
                macd = ma_delta
                repeat = 1

                qty = order_size(i, ma_delta)
            
                if abs(macd) >= 0.1:
                    qty = 10000
                    repeat = 2
         
                    
                #LONG ENTRY
                
                if ma_s[i] > ma_l[i] and macd > growth_threshold_up and stock_mid[i] > upper_band:
                    qty = order_size(i, ma_delta)
                    for j in range(repeat):
                        place_mkt(i, "BUY", qty)
                    last_order[i] = "BUY"

                # SHORT ENTRY
                elif ma_s[i] < ma_l[i] and macd < growth_threshold_down and stock_mid[i] < lower_band:
                    qty = order_size(i, ma_delta)
                    for j in range(repeat):
                        place_mkt(i, "SELL", qty)
                    last_order[i] = "SELL"


            if tick >= 13 and tick != previous_tick:
                trade(i)
                profit_take(i)
                
          
       
        previous_tick = tick
        sleep(0.1)
        tick, status = get_tick_status()
        ticks += 1

    # Keep the final chart on screen after loop ends
    plt.ioff()
    plt.show()


if __name__ == "__main__":
    main()
